\subsection{Degree sum formula}

\begin{frame}{Degrees of vertices}
\pause
  \begin{block}{Definition}
    The \textbf{degree} of a vertex is how many edges are coming out of it.
  \end{block}
\pause
  \begin{block}{Activity}
    Draw some examples of graphs and for each graph, do the following:
    \begin{itemize}
\pause
      \item  Count the vertices
\pause
      \item  Count the edges
\pause
      \item  Figure out the degrees of the vertices
\pause
      \item  Can you spot any relationships?
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Degree Sum Formula}
\pause
  \begin{block}{Theorem:}
    If we add up the degrees of all the vertices in the graph, the result is equal to twice the number of edges.
  \end{block}
\end{frame}

\begin{frame}{Degree Sum Formula}
\pause
  \begin{block}{Proof:}
\pause
    The degree of a vertex counts the number of edges coming out of it.\\
\pause
    The sum of all the degrees is the number of times any edge comes out of any vertex.\\
\pause
    Each edge comes out of exactly two vertices.\\
\pause
    Therefore the number of edges is twice the degree sum.
  \end{block}
\end{frame}

\begin{frame}{Handshaking Lemma}
\pause
  \begin{block}{Theorem:}
    The number of vertices which have an odd degree must be even.
  \end{block}
\pause
  \begin{block}{Folksy version:}
    At any business meeting, different people are shaking hands.\\
    The number of people who shook an odd number of other people's hands must be even.
  \end{block}
\end{frame}

\begin{frame}{Handshaking Lemma}
  \begin{block}{Proof:}
\pause
    The sum of the degrees must be even.\\
\pause
    If we add up a bunch of numbers
    $$a_1 + a_2 + \dots + a_n,$$
    the result is even if and only if an even amount of the $a_j$'s were odd.\\
\pause
    Therefore the number of odd-degree vertices must be even.
  \end{block}
\end{frame}

\subsection{Euler's Formula}
\begin{frame}{Connected graphs}
\pause
  \begin{block}{definition}
    A graph is called \textbf{connected} if you can travel from every vertex to every other by walking along a sequence of edges.
  \end{block}
  \begin{block}{definition}
    More generally, graphs are made up of \textbf{connected components}, ie. connected subgraphs which are as large as possible.
  \end{block}
\end{frame}

\begin{frame}{Planar graphs}
\pause
  \begin{block}{definition}
    a graph is called \textbf{planar} if none of the edges of the graph overlap.
  \end{block}
\pause
  \begin{block}{activity}
    draw some examples of connected planar graphs and for each graph, do the following:
    \begin{itemize}
      \item  count the vertices
      \item  count the edges
      \item  count the number of regions
      \item  can you spot any relationships?
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Euler's formula}
\pause
  \begin{block}{Theorem:}
    Let $E$ be the number of edges, $V$ the number of faces, and $F$ the number of divided regions of a connected planar graph (including the ``outside").  Then
  $$V-E+F = 2.$$
  \end{block}
\end{frame}

\begin{frame}{Euler's formula}
\pause
  \begin{block}{Proof:}
    Start with a graph.\\
\pause
    If it has an enclosed region, snip a boundary edge, creating a new graph.\\
\pause
    This merges the the region with an adjacent region, keeping the graph connected.\\
\pause
    Therefore the new graph has one less edge, one less region, but the same number of vertices.\\
\pause
    Therefore it doesn't change the value of $V-E+F$.\\
\pause
    Continue in this way until there's no enclosed regions, so $F=1$.
  \end{block}
\end{frame}

\begin{frame}{Euler's formula}
\pause
  \begin{block}{Proof (continued)}
\pause
    When $F=1$, the graph is a \textbf{tree}, connected without loops.\\
\pause
    It will have an vertex of degree $1$ at the tip of a branch.\\
\pause
    Remove that vertex and the edge it touched, ``pruning" the tree.\\
\pause
    This removes one edge and one vertex, leaving a smaller tree without changing the value of $V-E+F$.\\
\pause
    Prune until you are left with a single vertex and edge.\\
\pause
    You never changed the value of $V-E+F$, so $V-E+F=2$.
  \end{block}
\end{frame}

\begin{frame}{Multi-component Euler's formula}
\pause
  \begin{block}{Theorem:}
    Let $E$ be the number of edges, $V$ the number of faces, and $F$ the number of divided regions and $C$ the number of connected components of a planar graph (including the ``outside").  Then
  $$V-E+F = 1+C.$$
  \end{block}
\end{frame}
