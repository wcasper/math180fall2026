\subsection{Probability and Expectation Values}

\begin{frame}{Probability Intro}
  \begin{block}{Vocabulary}
    In \textbf{probability theory} we study \textbf{experiments}, like flipping a coin, rolling a die, or drawing a card.\\
    \begin{itemize}
\pause
      \item the set $\Omega$ of all possible outcomes is the \textbf{sample space}
\pause
      \item subsets $E\subseteq \Omega$ are called \textbf{events}
\pause
      \item the \textbf{probability} of an event $E$ is
      $$\mathbb{P}(E) = \frac{|E|}{|\Omega|}$$
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Probability Example}
  \begin{block}{Rolling a Die}
    Experiment: roll a six-sided die, the sample space is:
    $$\Omega = \{1,2,3,4,5,6\}$$
    \begin{itemize}
\pause
      \item the event of ``getting a 3" is $\{3\}$
      $$\mathbb{P}(\{3\}) = \frac{|\{3\}|}{|\Omega|} = \frac{1}{6}.$$
\pause
      \item the event of ``getting an even number" is $\{2,4,6\}$
      $$\mathbb{P}(\{2,4,6\}) = \frac{|\{2,4,6\}|}{|\Omega|} = \frac{3}{6} = \frac{1}{2}.$$
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Probability Example}
  \begin{block}{Drawing a Card}
    Experiment: draw a card from a standard deck
    \begin{itemize}
\pause
      \item what is the sample space?\\
\pause
        $\Omega = \{(2,\heartsuit),(3,\heartsuit),(4,\heartsuit),\dots,(K,\clubsuit),(A,\clubsuit)\}$
\pause
      \item what is the event "getting a face card in hearts"?\\
\pause
        $E = \{(J,\heartsuit),(Q,\heartsuit),(K,\heartsuit)\}$
\pause
      \item what is the probability of this event?\\
\pause
        $\mathbb{P}(E) = \frac{|E|}{|\Omega|} = \frac{3}{52}.$
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Probability Meaning}
  \begin{block}{Real-world interpretation}
    \begin{itemize}
\pause
      \item perform the same experiment a thousand times
\pause
      \item record how many times the outcome is in $E$
\pause
      \item fraction of times the outcome is in $E$ is approx. $\mathbb P(E)$
\pause
      \item more experiments -> closer to $\mathbb P(E)$
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Expectation Value}
  Start with a sample space
  $$\Omega = \{a_1,a_2,a_3,\dots, a_n\}.$$
\pause
  \begin{block}{Vocabulary}
    A \textbf{expectation value} of a \textbf{value function} $f(x)$ is
    $$\mathbb{E}(f) = f(a_1)\mathbb P(\{a_1\}) + f(a_2)\mathbb P(\{a_2\}) + \dots + f(a_n)\mathbb P(\{a_n\})$$
  \end{block}
\pause
  \begin{block}{Meaning}
    Represents the average value we obtain if we do thousands of experiments
  \end{block}
\end{frame}


\begin{frame}{Expectation Value Example}
  \begin{block}{Gambling with dice}
    Suppose we play a game where we roll a die.  If the outcome is a $6$, then we win $4$ dollars.  Otherwise, we lose a dollar.\\
    \begin{itemize}
\pause
      \item value function\\
      $f(1) = -1,\ f(2) = -1,\ f(3) = -1$\\
      $f(4) = -1,\ f(5) = -1,\ f(6) = 4$
\pause
      \item expectation value $\mathbb{E}(f)$\\
      $(-1)\cdot \frac{1}{6} + (-1)\cdot \frac{1}{6} + (-1)\cdot \frac{1}{6} + (-1)\cdot \frac{1}{6} + (-1)\cdot \frac{1}{6} + 4\cdot \frac{1}{6} = \frac{-1}{6}$
\pause
      \item if we play a thousand games, we'll lose an average of $17$ cents per game or about $166.67$ dollars total.
    \end{itemize}
  \end{block}
\end{frame}
